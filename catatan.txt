sudo certbot certonly --dns-digitalocean \
  --dns-digitalocean-credentials /etc/letsencrypt/digitalocean.ini \
  --dns-digitalocean-propagation-seconds 120 \
  -d mulaigym.id \
  -d *.mulaigym.id \
  --register-unsafely-without-email \
  --agree-tos \
  --preferred-chain "ISRG Root X1" \
  --force-renewal

sudo nano /etc/nginx/sites-available/mulaigym.conf
# HTTP → HTTPS redirect
server {
    listen 80;
    server_name mulaigym.id www.mulaigym.id;
    return 301 https://$host$request_uri;
}

# HTTPS proxy to Docker
server {
    listen 443 ssl http2;
    server_name mulaigym.id www.mulaigym.id;

    # SSL certificates from Certbot
    ssl_certificate /etc/letsencrypt/live/mulaigym.id/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/mulaigym.id/privkey.pem;
    
    # Security headers
    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;
    add_header X-Content-Type-Options "nosniff";
    add_header X-Frame-Options "SAMEORIGIN";

    # Proxy configuration
    location / {
        proxy_pass http://localhost:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        # Add these for better proxy handling
        proxy_redirect off;
        proxy_buffering off;
        proxy_read_timeout 1800;
        proxy_connect_timeout 1800;
    }

    # Let WhiteNoise handle all static files
    location /static/ {
        proxy_pass http://localhost:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        expires 1y;
        access_log off;
        add_header Cache-Control "public";
    }

    location /media/ {
        alias /root/mulai_web/media/;
        expires 1y;
        access_log off;
        add_header Cache-Control "public";
    }
}

sudo ln -s /etc/nginx/sites-available/mulaigym.conf /etc/nginx/sites-enabled/
sudo rm /etc/nginx/sites-enabled/default  # Remove default config
sudo nginx -t && sudo systemctl reload nginx

sudo crontab -e
0 3 * * * certbot renew --quiet --post-hook "docker restart mulai_web"
