# SSL Certificate Setup (Cloudflare DNS Challenge)
# Prerequisites: Domain nameservers pointing to Cloudflare

# Install Cloudflare DNS plugin
sudo apt update
sudo apt install python3-certbot-dns-cloudflare

# Create Cloudflare credentials file
sudo nano /etc/letsencrypt/cloudflare.ini
# Add: dns_cloudflare_api_token = YOUR_CLOUDFLARE_API_TOKEN
# Get API token from: Cloudflare Dashboard → My Profile → API Tokens
# Permissions: Zone:DNS:Edit, Zone:Zone:Read for mulaigym.id

# Secure credentials file
sudo chmod 600 /etc/letsencrypt/cloudflare.ini

# Obtain SSL certificate (supports wildcards)
sudo certbot certonly --dns-cloudflare \
  --dns-cloudflare-credentials /etc/letsencrypt/cloudflare.ini \
  --dns-cloudflare-propagation-seconds 60 \
  -d mulaigym.id \
  -d *.mulaigym.id \
  --register-unsafely-without-email \
  --agree-tos \
  --preferred-chain "ISRG Root X1" \
  --force-renewal

# Nginx Configuration
sudo nano /etc/nginx/sites-available/mulaigym.conf

# HTTP → HTTPS redirect
server {
    listen 80;
    server_name mulaigym.id www.mulaigym.id;
    return 301 https://$host$request_uri;
}

# HTTPS proxy to Docker
server {
    listen 443 ssl http2;
    server_name mulaigym.id www.mulaigym.id;

    # SSL certificates from Certbot
    ssl_certificate /etc/letsencrypt/live/mulaigym.id/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/mulaigym.id/privkey.pem;
    
    # Security headers
    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;
    add_header X-Content-Type-Options "nosniff";
    add_header X-Frame-Options "SAMEORIGIN";

    # Proxy configuration
    location / {
        proxy_pass http://localhost:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        # Add these for better proxy handling
        proxy_redirect off;
        proxy_buffering off;
        proxy_read_timeout 1800;
        proxy_connect_timeout 1800;
    }

    # Let WhiteNoise handle all static files
    location /static/ {
        proxy_pass http://localhost:8000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        expires 1y;
        access_log off;
        add_header Cache-Control "public";
    }

    location /media/ {
        alias /root/mulai_web/media/;
        expires 1y;
        access_log off;
        add_header Cache-Control "public";
    }
}

# Enable site and restart nginx
sudo ln -s /etc/nginx/sites-available/mulaigym.conf /etc/nginx/sites-enabled/
sudo rm /etc/nginx/sites-enabled/default  # Remove default config
sudo nginx -t && sudo systemctl reload nginx

# Auto-renewal cron job (runs daily at 3 AM)
sudo crontab -e
0 3 * * * certbot renew --quiet --post-hook "systemctl reload nginx && docker restart mulai_web"

# Create Django superuser
docker exec -e DJANGO_SUPERUSER_USERNAME=admin \
           -e DJANGO_SUPERUSER_EMAIL=admin@example.com \
           -e DJANGO_SUPERUSER_PASSWORD=your_secure_password \
           mulai_web python manage.py createsuperuser --noinput

# Useful Commands
# Check certificate status: sudo certbot certificates
# Manual renewal: sudo certbot renew --force-renewal
# Test auto-renewal: sudo certbot renew --dry-run
# Restart services: sudo systemctl reload nginx && docker restart mulai_web
# Check Docker logs: docker logs mulai_web

# Troubleshooting
# Error 526: Usually expired certificate or DNS mismatch
# Check nginx config: sudo nginx -t
# Check certificate expiry: sudo certbot certificates
# View renewal logs: sudo tail -50 /var/log/letsencrypt/letsencrypt.log
